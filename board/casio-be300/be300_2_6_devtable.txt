# BE-300 Linux 2.6.8.1 + Opie JFFS2 device table for mkfs.jffs2 --devtable.
#
# Linux 2.6.8.1 has no devtmpfs, so every device node Opie/BusyBox will open
# at runtime must exist in the JFFS2 image.  mkfs.jffs2 reads this file and
# bakes the nodes into the image even when running inside an unprivileged
# Docker container (where mknod fails).
#
# Format:
#   <name> <type> <mode> <uid> <gid> <major> <minor> <start> <inc> <count>
#
# - type: c = char dev, b = block dev, d = directory
# - For singletons leave <start> <inc> <count> as `-`.
# - For ranges, <count> nodes get the suffix `<start>`, `<start>+<inc>`, ...,
#   appended to <name> (with the major/minor incremented by <inc> each step).
#
# Majors/minors cross-checked against include/linux/major.h, the
# patches/linux-2.6.8.1-be300 drivers and the 2.4.18 devtable.

/dev            d 755 0 0 -  -  -  -  -
/dev/input      d 755 0 0 -  -  -  -  -
/dev/pts        d 755 0 0 -  -  -  -  -

# core console + nullables
/dev/console    c 600 0 0 5  1  -  -  -
/dev/null       c 666 0 0 1  3  -  -  -
/dev/zero       c 666 0 0 1  5  -  -  -
/dev/full       c 666 0 0 1  7  -  -  -
/dev/random     c 644 0 0 1  8  -  -  -
/dev/urandom    c 644 0 0 1  9  -  -  -
/dev/mem        c 640 0 0 1  1  -  -  -
/dev/kmem       c 640 0 0 1  2  -  -  -
/dev/kmsg       c 644 0 0 1  11 -  -  -

# virtual terminals (BusyBox init opens /dev/tty0; Opie's be300-vtmode
# switches it to KD_GRAPHICS).
/dev/tty        c 666 0 0 5  0  -  -  -
/dev/tty        c 600 0 0 4  0  0  1  4

# Dock UART: drivers/char/serial_be300.c registers ttyS0 as TTY_MAJOR (4),
# minor 64 -- the same node as the 2.4.18 kernel.
/dev/ttyS0      c 660 0 0 4  64 -  -  -

# UNIX98 ptys (embeddedkonsole opens /dev/ptmx; pts/N appear once devpts
# is mounted by /etc/init.d/rcS).
/dev/ptmx       c 666 0 0 5  2  -  -  -

# framebuffer
/dev/fb0        c 660 0 0 29 0  -  -  -

# evdev nodes.  The board Makefile links stowaway.o before touch.o and the
# buttons register at late_initcall time, so evdev minors are fixed:
# event0 = Stowaway keyboard, event1 = PIU touch panel, event2 = buttons
# (Opie reads QWS_MOUSE_PROTO=TPanel:/dev/input/event1,
# QWS_KEYBOARD=USB:/dev/input/event0).
/dev/input/event c 660 0 0 13 64 0  1  4

# MTD raw + block (rootfs lives on mtdblock3; the first three partitions
# expose the WinCE table, SPL, and vmlinux).
/dev/mtd        c 600 0 0 90 0  0  2  4
/dev/mtdblock   b 600 0 0 31 0  0  1  4
